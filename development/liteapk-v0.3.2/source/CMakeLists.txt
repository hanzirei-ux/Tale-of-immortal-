cmake_minimum_required(VERSION 3.22)
project(toi_modmenu VERSION 0.3.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(toi_core STATIC native/bridge.cpp native/sha256.cpp native/il2cpp_backend.cpp native/modifiers.cpp)
set_target_properties(toi_core PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
target_include_directories(toi_core PUBLIC native)
target_compile_options(toi_core PRIVATE -Wall -Wextra -Wpedantic -Werror)
if(ANDROID)
  if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
    message(FATAL_ERROR "Only the ARM64 baseline has been researched")
  endif()
  if(NOT DEFINED TOI_DOBBY_ROOT)
    set(TOI_DOBBY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party/Dobby")
  endif()
  set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
  set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
  set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
  add_subdirectory(${TOI_DOBBY_ROOT} ${CMAKE_CURRENT_BINARY_DIR}/dobby)
  add_library(toimodmenu SHARED native/jni.cpp native/mod_api.cpp native/game_runtime.cpp native/managed_access.cpp native/game_editors.cpp native/localization_cache.cpp native/control_gateway.cpp)
  set_target_properties(toimodmenu PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
  target_include_directories(toimodmenu PRIVATE ${TOI_DOBBY_ROOT}/include)
  target_compile_options(toimodmenu PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(TARGET dobby_static)
    target_link_libraries(toimodmenu PRIVATE toi_core dobby_static dl log)
  else()
    target_link_libraries(toimodmenu PRIVATE toi_core dobby dl log)
  endif()
  target_link_options(toimodmenu PRIVATE -Wl,--exclude-libs,ALL,-z,relro,-z,now,-z,noexecstack,-z,max-page-size=16384)
else()
  find_package(Threads REQUIRED)
  add_executable(toi_core_tests tests/native_tests.cpp)
  target_link_libraries(toi_core_tests PRIVATE toi_core dl Threads::Threads)
  enable_testing()
  add_test(NAME core COMMAND toi_core_tests)
  add_executable(toi_policy_tests tests/policy_tests.cpp native/mod_api.cpp)
  target_link_libraries(toi_policy_tests PRIVATE toi_core dl Threads::Threads)
  add_test(NAME policy COMMAND toi_policy_tests)
  add_executable(toi_editor_tests tests/editor_tests.cpp native/managed_access.cpp native/game_editors.cpp native/localization_cache.cpp native/control_gateway.cpp)
  target_include_directories(toi_editor_tests PRIVATE native)
  target_compile_options(toi_editor_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(toi_editor_tests PRIVATE toi_core Threads::Threads)
  add_test(NAME editors COMMAND toi_editor_tests)
  add_executable(toi_hotfix_tests tests/hotfix_tests.cpp native/control_gateway.cpp native/localization_cache.cpp native/managed_access.cpp)
  target_compile_options(toi_hotfix_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
  target_link_libraries(toi_hotfix_tests PRIVATE toi_core Threads::Threads)
  add_test(NAME hotfix COMMAND toi_hotfix_tests)
endif()
